# STATIC-ONLY v3 — статика + редиректы старых лендингов
Options -Indexes
DirectoryIndex index.html
AddDefaultCharset UTF-8

<IfModule mod_mime.c>
  AddType text/html; charset=utf-8 .html
  AddType application/xml; charset=utf-8 .xml
  AddType application/javascript; charset=utf-8 .js
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.(html)$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
  <FilesMatch "\.(css|js|svg|png|jpg|jpeg|webp|woff2?|ico)$">
    Header set Cache-Control "public, max-age=2592000"
  </FilesMatch>
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json application/xml image/svg+xml
</IfModule>

<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /

# 301 со старых лендингов на разделы услуг (вес и индекс перетекают)
RewriteRule ^parser-2gis/?$ /services/parser-2gis/ [R=301,L]
RewriteRule ^parser-cen/?$ /services/parser-cen/ [R=301,L]
RewriteRule ^lampa-parser-ne-otvechaet/?$ /services/support/ [R=301,L]

# технический хвост прежних версий
RewriteRule ^index1\.html$ / [R=301,L]
RewriteRule ^index\.php$ / [R=301,L]
RewriteRule ^xmlrpc\.php$ / [R=301,L]
RewriteRule ^_wp-trash - [F,L]
RewriteRule ^constructor - [F,L]

# реально существующие файлы и папки отдаём как есть
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]
</IfModule>

ErrorDocument 404 /index.html
